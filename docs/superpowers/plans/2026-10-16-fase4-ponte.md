```csharp
using System;
using System.IO;
using System.Linq;
using System.Text.RegularExpressions;
using Orbit.Services;

namespace Orbit.Bridge;

public enum MediaCommand { PlayPause, Next, Prev }

// Regras puras da ponte (sem rede nem disco) — testadas no harness.
public static class BridgeRules
{
    public const long MaxUploadBytes = 500L * 1024 * 1024;
    public const int MaxClipboardChars = 100_000;
    private const int MaxFileName = 120;

    private static readonly Regex Spaces = new(@"\s+");
    private static readonly string[] Reserved =
        { "CON", "PRN", "AUX", "NUL", "COM1", "COM2", "COM3", "COM4", "COM5", "COM6", "COM7", "COM8", "COM9",
          "LPT1", "LPT2", "LPT3", "LPT4", "LPT5", "LPT6", "LPT7", "LPT8", "LPT9" };

    // Só http/https: nada de javascript:, file:, ms-settings: etc. vindo do celular.
    public static bool IsOpenableUrl(string? url) =>
        Uri.TryCreate(url?.Trim(), UriKind.Absolute, out var u)
        && (u.Scheme == Uri.UriSchemeHttp || u.Scheme == Uri.UriSchemeHttps);

    public static string Host(string url) => Uri.TryCreate(url.Trim(), UriKind.Absolute, out var u) ? u.Host : "";

    public static string? BearerToken(string? header) =>
        header is not null && header.StartsWith("Bearer ", StringComparison.OrdinalIgnoreCase)
            ? header[7..].Trim()
            : null;

    // Nome de arquivo vindo do celular: sem pastas (path traversal), sem caracteres inválidos,
    // sem nomes reservados, com tamanho limitado.
    public static string SafeFileName(string? name)
    {
        var n = Path.GetFileName((name ?? "").Replace('/', '\\'));
        foreach (var c in Path.GetInvalidFileNameChars()) n = n.Replace(c, '_');
        n = n.Trim().TrimEnd('.', ' ');
        if (n.Length == 0 || n.All(ch => ch == '.')) n = "arquivo";
        if (Reserved.Contains(Path.GetFileNameWithoutExtension(n).ToUpperInvariant())) n = "_" + n;
        if (n.Length > MaxFileName)
        {
            var ext = Path.GetExtension(n);
            n = n[..(MaxFileName - ext.Length)] + ext;
        }
        return n;
    }

    // Não sobrescreve: "foto.jpg" → "foto (1).jpg" → "foto (2).jpg"…
    public static string UniqueName(string dir, string name, Func<string, bool> exists)
    {
        if (!exists(Path.Combine(dir, name))) return name;
        var stem = Path.GetFileNameWithoutExtension(name);
        var ext = Path.GetExtension(name);
        for (int i = 1; ; i++)
        {
            var candidate = $"{stem} ({i}){ext}";
            if (!exists(Path.Combine(dir, candidate))) return candidate;
        }
    }

    public static PowerAction? ParsePower(string action) => action.ToLowerInvariant() switch
    {
        "lock" => PowerAction.Bloquear,
        "sleep" => PowerAction.Suspender,
        "restart" => PowerAction.Reiniciar,
        "shutdown" => PowerAction.Desligar,
        _ => null,
    };

    public static MediaCommand? ParseMedia(string command) => command.ToLowerInvariant() switch
    {
        "playpause" => MediaCommand.PlayPause,
        "next" => MediaCommand.Next,
        "prev" => MediaCommand.Prev,
        _ => null,
    };

    // Leituras que o PWA faz em polling (stats, mídia) não poluem a auditoria.
    public static bool ShouldAudit(string method, string path) =>
        !(string.Equals(method, "GET", StringComparison.OrdinalIgnoreCase)
          && (path.EndsWith("/stats", StringComparison.OrdinalIgnoreCase) || path.EndsWith("/media", StringComparison.OrdinalIgnoreCase)));

    public static string Preview(string text)
    {
        var s = Spaces.Replace(text, " ").Trim();
        return s.Length > 60 ? s[..60] + "…" : s;
    }

    public static string NormalizeBaseUrl(string? url)
    {
        var s = (url ?? "").Trim().TrimEnd('/');
        if (s.Length == 0) return "";
        return s.Contains("://", StringComparison.Ordinal) ? s : "https://" + s;
    }

    // O token vai no fragmento (#): o navegador não o manda em nenhuma requisição.
    public static string PairingLink(string baseUrl, string token) =>
        NormalizeBaseUrl(baseUrl) + "/#token=" + Uri.EscapeDataString(token);
}
```

```csharp
using System;
using System.Buffers.Text;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using Orbit.Services;

namespace Orbit.Bridge;

public sealed record TokenInfo(string Name, string Hash, DateTime CreatedAt);

// Tokens da ponte. O arquivo guarda só o SHA-256 (o token em si aparece uma vez, no pareamento).
// Verify roda em threads do Kestrel; Create/Revoke na UI: tudo sob lock.
public sealed class TokenStore
{
    public static readonly string FilePath = Path.Combine(SettingsStore.Dir, "tokens.json");

    private static readonly JsonSerializerOptions WriteOptions = new() { WriteIndented = true };
    private readonly string _path;
    private readonly object _gate = new();
    private readonly List<TokenInfo> _tokens;

    public TokenStore(string path)
    {
        _path = path;
        _tokens = Load(path);
    }

    public IReadOnlyList<TokenInfo> List()
    {
        lock (_gate) return _tokens.ToList();
    }

    // Devolve o token em texto (mostrar uma vez só).
    public string Create(string name)
    {
        var token = Base64Url.EncodeToString(RandomNumberGenerator.GetBytes(32));
        lock (_gate)
        {
            _tokens.Add(new TokenInfo(name, Hash(token), DateTime.Now));
            Save();
        }
        return token;
    }

    // Nome do token, ou null. Comparação em tempo constante.
    public string? Verify(string? token)
    {
        if (string.IsNullOrEmpty(token)) return null;
        var candidate = Convert.FromHexString(Hash(token));
        lock (_gate)
        {
            foreach (var t in _tokens)
                if (CryptographicOperations.FixedTimeEquals(candidate, Convert.FromHexString(t.Hash)))
                    return t.Name;
        }
        return null;
    }

    public void Revoke(string hash)
    {
        lock (_gate)
        {
            _tokens.RemoveAll(t => t.Hash == hash);
            Save();
        }
    }

    public static string Hash(string token) => Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(token)));

    private static List<TokenInfo> Load(string path)
    {
        try
        {
            if (!File.Exists(path)) return new();
            return JsonSerializer.Deserialize<List<TokenInfo>>(File.ReadAllText(path))?
                .Where(t => t.Hash is { Length: 64 })
                .ToList() ?? new();
        }
        catch
        {
            return new(); // corrompido → nenhum token (o usuário pareia de novo)
        }
    }

    private void Save()
    {
        try
        {
            Directory.CreateDirectory(Path.GetDirectoryName(_path)!);
            var tmp = _path + ".tmp";
            File.WriteAllText(tmp, JsonSerializer.Serialize(_tokens, WriteOptions));
            File.Move(tmp, _path, overwrite: true);
        }
        catch
        {
            // sem disco: o token vale até fechar o app
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.Json;
using Orbit.Services;

namespace Orbit.Bridge;

public sealed record AuditEntry(DateTime When, string Token, string Action, string Result);

// Uma linha JSON por ação em audit.jsonl + as últimas N em memória (aba iPhone).
// Write roda em threads do Kestrel: tudo sob lock; Written dispara fora da UI thread.
public sealed class AuditLog
{
    public static readonly string FilePath = Path.Combine(SettingsStore.Dir, "audit.jsonl");
    private const int RecentCount = 30;
    private const long MaxBytes = 5 * 1024 * 1024; // ponytail: rotação simples (1 arquivo .old)

    private readonly string _path;
    private readonly object _gate = new();
    private readonly LinkedList<AuditEntry> _recent = new();

    public event Action<AuditEntry>? Written;

    public AuditLog(string path) => _path = path;

    public void Write(string token, string action, string result)
    {
        var entry = new AuditEntry(DateTime.Now, token, action, result);
        lock (_gate)
        {
            try
            {
                Directory.CreateDirectory(Path.GetDirectoryName(_path)!);
                if (File.Exists(_path) && new FileInfo(_path).Length > MaxBytes)
                    File.Move(_path, _path + ".old", overwrite: true);
                File.AppendAllText(_path, FormatLine(entry) + "\n");
            }
            catch
            {
                // sem disco: fica só em memória
            }
            _recent.AddFirst(entry);
            if (_recent.Count > RecentCount) _recent.RemoveLast();
        }
        Written?.Invoke(entry);
    }

    public IReadOnlyList<AuditEntry> RecentEntries()
    {
        lock (_gate) return _recent.ToList();
    }

    public static string FormatLine(AuditEntry e) => JsonSerializer.Serialize(new
    {
        quando = e.When.ToString("o"),
        token = e.Token,
        acao = e.Action,
        resultado = e.Result,
    });
}
```

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Net;
using System.Threading.RateLimiting;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Hosting;
using Microsoft.AspNetCore.Hosting.Server;
using Microsoft.AspNetCore.Hosting.Server.Features;
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Http.Features;
using Microsoft.AspNetCore.RateLimiting;
using Microsoft.AspNetCore.Routing;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using Serilog;

namespace Orbit.Bridge;

public sealed record LinkRequest(string? Url);
public sealed record TextRequest(string? Text);
public sealed record VolumeRequest(double Percent);

// Kestrel dentro do processo do Orbit, só em 127.0.0.1 (o `tailscale serve` é quem expõe).
// Toda rota /api exige token e passa por rate limit; /health e o PWA (arquivos estáticos) não.
public sealed class BridgeServer : IAsyncDisposable
{
    private WebApplication? _app;

    public int Port { get; private set; }
    public bool Running => _app is not null;

    // port 0 = porta efêmera (testes).
    public async Task StartAsync(int port, string webRoot, TokenStore tokens, AuditLog audit, BridgeHandlers h)
    {
        if (_app is not null) return;
        bool hasWeb = Directory.Exists(webRoot);
        var builder = WebApplication.CreateSlimBuilder(new WebApplicationOptions
        {
            ContentRootPath = AppContext.BaseDirectory,
            WebRootPath = hasWeb ? webRoot : null,
        });
        builder.Logging.ClearProviders();
        builder.WebHost.ConfigureKestrel(k =>
        {
            k.Listen(IPAddress.Loopback, port); // nunca 0.0.0.0 (RNF01)
            k.Limits.MaxRequestBodySize = BridgeRules.MaxUploadBytes;
        });
        builder.Services.Configure<FormOptions>(o => o.MultipartBodyLengthLimit = BridgeRules.MaxUploadBytes);
        builder.Services.AddRateLimiter(o =>
        {
            o.RejectionStatusCode = StatusCodes.Status429TooManyRequests;
            o.AddFixedWindowLimiter("api", w =>
            {
                w.PermitLimit = 120;
                w.Window = TimeSpan.FromMinutes(1);
                w.QueueLimit = 0;
            });
        });

        var app = builder.Build();
        app.UseRateLimiter();
        if (hasWeb)
        {
            app.UseDefaultFiles();
            app.UseStaticFiles();
        }
        app.MapGet("/health", () => Results.Ok(new { ok = true }));

        var api = app.MapGroup("/api").RequireRateLimiting("api");
        api.AddEndpointFilter(async (ctx, next) =>
        {
            var req = ctx.HttpContext.Request;
            var action = $"{req.Method} {req.Path}";
            var name = tokens.Verify(BridgeRules.BearerToken(req.Headers.Authorization.ToString()));
            if (name is null)
            {
                audit.Write("?", action, "401");
                return Results.Unauthorized();
            }
            try
            {
                var result = await next(ctx);
                if (BridgeRules.ShouldAudit(req.Method, req.Path))
                    audit.Write(name, action, ((result as IStatusCodeHttpResult)?.StatusCode ?? 200).ToString());
                return result;
            }
            catch (Exception ex) when (ex is not BadHttpRequestException)
            {
                Log.Warning(ex, "Ponte: falha em {Acao}", action);
                audit.Write(name, action, "erro");
                return Results.Problem("Falha no PC.", statusCode: StatusCodes.Status500InternalServerError);
            }
        });
        MapApi(api, h);

        await app.StartAsync();
        _app = app;
        Port = ActualPort(app) ?? port;
        Log.Information("Ponte ouvindo em 127.0.0.1:{Porta}", Port);
    }

    private static void MapApi(RouteGroupBuilder api, BridgeHandlers h)
    {
        api.MapGet("/stats", () => h.GetStats());

        api.MapPost("/links/open", async (LinkRequest req) =>
        {
            if (!BridgeRules.IsOpenableUrl(req.Url)) return Results.BadRequest(new { erro = "Só links http/https." });
            var url = req.Url!.Trim();
            await h.OpenUrl(url);
            h.Notify(new BridgeEvent("Link aberto no PC", BridgeRules.Host(url)));
            return Results.NoContent();
        });

        api.MapGet("/clipboard", async () => Results.Ok(new { text = await h.GetClipboard() }));
        api.MapPut("/clipboard", async (TextRequest req) =>
        {
            if (req.Text is null || req.Text.Length > BridgeRules.MaxClipboardChars)
                return Results.BadRequest(new { erro = "Texto vazio ou grande demais." });
            await h.SetClipboard(req.Text);
            h.Notify(new BridgeEvent("Texto no clipboard do PC", BridgeRules.Preview(req.Text)));
            return Results.NoContent();
        });

        api.MapPost("/power/{action}", async (string action) =>
        {
            if (BridgeRules.ParsePower(action) is not { } a) return Results.NotFound();
            await h.Power(a);
            return Results.NoContent();
        });

        api.MapGet("/media", async () => Results.Ok(await h.GetMedia()));
        api.MapPost("/media/{command}", async (string command) =>
        {
            if (BridgeRules.ParseMedia(command) is not { } m) return Results.NotFound();
            await h.Media(m);
            return Results.NoContent();
        });
        api.MapPut("/media/volume", async (VolumeRequest req) =>
        {
            await h.SetVolume(Math.Clamp(req.Percent, 0, 100));
            return Results.NoContent();
        });

        api.MapPost("/files", async (HttpRequest request) =>
        {
            if (!request.HasFormContentType) return Results.BadRequest(new { erro = "Envie multipart/form-data." });
            var form = await request.ReadFormAsync();
            if (form.Files.Count == 0) return Results.BadRequest(new { erro = "Nenhum arquivo." });
            var dir = h.UploadFolder();
            Directory.CreateDirectory(dir);
            var saved = new List<string>();
            foreach (var file in form.Files)
            {
                var name = BridgeRules.UniqueName(dir, BridgeRules.SafeFileName(file.FileName), File.Exists);
                await using (var target = File.Create(Path.Combine(dir, name)))
                    await file.CopyToAsync(target);
                saved.Add(name);
            }
            h.Notify(new BridgeEvent(saved.Count == 1 ? "Arquivo do iPhone" : $"{saved.Count} arquivos do iPhone",
                string.Join(", ", saved)));
            return Results.Ok(new { saved });
        });
    }

    private static int? ActualPort(WebApplication app)
    {
        var address = app.Services.GetRequiredService<IServer>().Features.Get<IServerAddressesFeature>()?.Addresses.FirstOrDefault();
        return address is not null && Uri.TryCreate(address, UriKind.Absolute, out var u) ? u.Port : null;
    }

    public async Task StopAsync()
    {
        if (_app is null) return;
        var app = _app;
        _app = null;
        await app.StopAsync();
        await app.DisposeAsync();
        Log.Information("Ponte desligada");
    }

    public ValueTask DisposeAsync() => new(StopAsync());
}
```

```csharp
            bmp.StreamSource = new MemoryStream(png);
            bmp.EndInit();
            bmp.Freeze();
            return bmp;
        }
        catch (Exception ex)
        {
            Log.Warning(ex, "Ponte: QR indisponível; mostrando só o link");
```

```js
// Rede primeiro (atualizações chegam), cache como plano B offline. /api e /health nunca em cache.
const CACHE = 'orbit-v1';
const SHELL = ['./', './index.html', './app.js', './style.css', './manifest.webmanifest',
  './icon-192.png', './icon-512.png', './apple-touch-icon.png'];

  e.waitUntil(caches.open(CACHE).then(c => c.addAll(SHELL)).then(() => self.skipWaiting())));

self.addEventListener('activate', e =>
  e.waitUntil(caches.keys()
    .then(keys => Promise.all(keys.filter(k => k !== CACHE).map(k => caches.delete(k))))
    .then(() => self.clients.claim())));

self.addEventListener('fetch', e => {
  const url = new URL(e.request.url);
  if (e.request.method !== 'GET' || url.pathname.startsWith('/api') || url.pathname === '/health') return;
  e.respondWith(
    fetch(e.request)
      .then(r => { const copy = r.clone(); caches.open(CACHE).then(c => c.put(e.request, copy)); return r; })
      .catch(() => caches.match(e.request).then(r => r || caches.match('./index.html'))));
});
```
